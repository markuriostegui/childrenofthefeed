// Set up a MutationObserver to watch for the dynamic input
const searchInputContainer = document.getElementById('js-header_search_input_sub_container');
const observer = new MutationObserver((mutations) => {
    mutations.forEach((mutation) => {
        if (mutation.addedNodes.length) {
            const searchInput = document.getElementById('header_search_input_sub');
            if (searchInput) {
                initializeSearchBehavior(searchInput);
                observer.disconnect();
            }
        }
    });
});

observer.observe(searchInputContainer, {
    childList: true,
    subtree: true
});

function initializeSearchBehavior(searchInput) {
    const searchContainer = document.getElementById('header_search_form_sub');
    const subnavWrapper = document.querySelector('.subnav-content-wrapper');
    const headerWrapper = document.querySelector('.archive-header-container.the-sub-nav');
    const bodyWrapper = document.querySelector('body');

    searchInput.addEventListener('focus', function() {
        // Manually trigger the searchfocus event to load typeahead terms
        window.dispatchEvent(new Event('searchfocus'));
        
        searchContainer.classList.add('search-focused');
        subnavWrapper.classList.add('search-focused');
        headerWrapper.classList.add('search-focused');
        bodyWrapper.classList.add('mega-open');
        searchContainer.classList.add('search-transitioning');
        setTimeout(() => {
            searchContainer.classList.remove('search-transitioning');
        }, 300);
    });

    searchInput.addEventListener('blur', function() {
        // Only collapse if there's no search query
        if (!this.value) {
            searchContainer.classList.remove('search-focused');
            subnavWrapper.classList.remove('search-focused');
            headerWrapper.classList.remove('search-focused');
            bodyWrapper.classList.remove('mega-open');
            searchContainer.classList.add('search-transitioning');
            setTimeout(() => {
                searchContainer.classList.remove('search-transitioning');

document.addEventListener('DOMContentLoaded', function () {
    const archiveHeader = document.querySelector('#archive_header');
    const firstArticle = document.querySelector('div.highlights-container');
    const header = document.querySelector('header#masthead');

    const showElements = () => {
        if (!archiveHeader) {
            return;
        }

        archiveHeader.style.height = '110px';
        archiveHeader.style.opacity = '1';
    };

    const hideElements = () => {
        if (!archiveHeader) {
            return;
        }

        archiveHeader.style.height = '0';
        archiveHeader.style.opacity = '0';
    };

    const checkScrollPosition = () => {
        if (!firstArticle) {
            showElements();
            return;
        }

        const firstArticleTop = firstArticle.getBoundingClientRect().top + window.scrollY;
        const firstArticleBottom = firstArticle.getBoundingClientRect().bottom + window.scrollY;
        const currentScroll = window.scrollY;
        const headerHasActiveClass = header && header.classList.contains('uk-active');

        if (currentScroll < firstArticleTop - 230 || !headerHasActiveClass) {
            showElements();
        } else if (currentScroll > firstArticleBottom) {
          hideElements();
        } else {
            hideElements();
        }
    };

    checkScrollPosition();

    window.addEventListener('scroll', checkScrollPosition);

    if (header) {
        const observer = new MutationObserver(function(mutations) {
            mutations.forEach(function(mutation) {
                if (mutation.type === 'attributes' && mutation.attributeName === 'class') {
                    checkScrollPosition();
                }
            });
        });

        observer.observe(header, {
            attributes: true,
            attributeFilter: ['class']
        });
    }
});

	function setHeroOffset() {
		
		var isMobile = ( $('#menu-toggle').is(':visible') );
		var $header = ( isMobile ) ? $('#branding-column') : $('#masthead');
		
		if ( $header.length < 1 ) {
			console.warn('Header not found.');
			return;
		}
		
		var headerH = parseInt( $header.children().first().outerHeight() )
		
		if ( isNaN(headerH) || headerH === 0 ) {
			console.warn('Header does not have a height.');
			return;
		}
		
		if ( headerH > sanityCheckHeight ) {
			console.warn('Header too tall. Setting default.');
			headerH = ( isMobile ) ? 59 : 90;
		}

		var top = headerH;
		
		// Get header top position relative to document. Using placeholder since sticky (fixed pos) header will have top offset of 0
		var headerOffset = $('#js-masthead-position').offset();
		
		// Header not at top of document? (e.g. due to an alert bar)
		if ( typeof headerOffset !== 'undefined' && typeof headerOffset.top === 'number' && headerOffset.top > 0 ) {
		
			var scrollPosTop = parseInt( $(window).scrollTop() );
			var stickyAdjustment = headerOffset.top - scrollPosTop;
			
			// ...and user hasn't entirely scrolled past site header?
			if ( (stickyAdjustment + headerH) > 0 ) {
			
				// Add vertical space above header to mega bg position
				top += stickyAdjustment;
				
			}
			
		}

		// Hero adjustments
		$styles.html('.company-hero-row-1, .has-hero-bg .hero-section { margin-top: -' + parseFloat(top) + 'px; } .has-hero-overhang .hero-section .overhang-wrapper { margin-bottom: -' + parseFloat(top) + 'px; } .company-hero-row-1, .has-hero-overhang .hero-section + .uk-section, .has-hero-bg .hero-section { padding-top: ' + parseFloat(top) + 'px; } .has-hero-bg .hero-section { padding-bottom: ' + parseFloat(top) + 'px; } .hero-intro-container { margin-bottom: ' + parseFloat(top) + 'px; }');
		
		// Nav/header adjustments
		if ( typeof stickyAdjustment === 'number' && (stickyAdjustment + headerH) > 0 ) {
			$styles.append('@media screen and (max-width: 1160px) { .main-nav-container { top: ' + parseFloat(top) + 'px; } .tray-open .site-header { top: ' + parseFloat(stickyAdjustment) + 'px; } }');

		}

			stylesAdded = true;
		}

		function analyzeImage(img, callback) {
			if (!img || typeof callback !== 'function') {
				return;
			}
			if (luminanceCache && luminanceCache.has(img)) {
				callback(luminanceCache.get(img));
				return;
			}

			function compute() {
				var width = img.naturalWidth || img.width;
				var height = img.naturalHeight || img.height;
				if (!width || !height) {
					callback(null);
					return;
				}
				var sampleSize = 24;
				var sw = Math.min(sampleSize, width);
				var sh = Math.min(sampleSize, height);
				var canvas = document.createElement('canvas');
				var ctx = canvas.getContext && canvas.getContext('2d');
				if (!ctx) {
					callback(null);
					return;
				}
				canvas.width = sw;
				canvas.height = sh;
				var positions = [
					{ x: Math.max(0, width - sw), y: 0 },
					{ x: Math.max(0, width - sw), y: Math.max(0, Math.round(height / 2 - sh / 2)) },
					{ x: Math.max(0, width - sw), y: Math.max(0, height - sh) },
				];
				try {
					var samples = [];
					for (var i = 0; i < positions.length; i++) {
						var pos = positions[i];
						ctx.clearRect(0, 0, sw, sh);
						ctx.drawImage(img, pos.x, pos.y, sw, sh, 0, 0, sw, sh);
						var data = ctx.getImageData(0, 0, sw, sh).data;
						var total = 0;
						var pixels = data.length / 4;
						for (var j = 0; j < data.length; j += 4) {
							var r = data[j];
							var g = data[j + 1];
							var b = data[j + 2];
							total += 0.2126 * r + 0.7152 * g + 0.0722 * b;
						}
						samples.push(total / pixels);
					}
					var darkThreshold = 125;
					var darkCount = samples.filter(function (val) {
						return val <= darkThreshold;
					}).length;
					var theme = darkCount >= 2 ? 'light' : 'dark';
					if (luminanceCache) {
						luminanceCache.set(img, theme);
					}
					callback(theme);
				} catch (err) {
					callback(null);
				}
			}

			if (!img.complete || !img.naturalWidth) {
				img.addEventListener('load', function onLoad() {
					img.removeEventListener('load', onLoad);
					compute();
				});
				return;
			}

			compute();
		}

		function getVisibleSlideImage(slideshow) {
			if (!slideshow) {
				return null;
			}
			var selectors = ['.jetpack-slideshow-slide img', '.jetpack-slideshow-window img', 'img'];
			var imgs = null;
			for (var i = 0; i < selectors.length; i++) {
				var found = slideshow.querySelectorAll(selectors[i]);
				if (found && found.length) {
					imgs = found;
					break;
				}
			}
			if (!imgs || !imgs.length) {
				return null;
			}
			var fallback = imgs[0];
			for (var j = 0; j < imgs.length; j++) {
				var img = imgs[j];
				var holder = img.closest('.jetpack-slideshow-slide') || img;
				var style = window.getComputedStyle(holder);
				if (!style) {
					continue;

					continue;

				if (opacity < 0.1) {
					continue;
				}
				return img;
			}
			return fallback;
		}

		function watchSlideshow(slideshow) {

		}

				if ($('.gdprconsent-button').is(':visible')) {
					// Hide the div if it's visible
					$('.gdprconsent-button button').click();
				}
			});

			// Add a click event listener to the document
			$(document).click(function(event) {
				// Check if the clicked element is not inside the div
				if (!$(event.target).closest('.gdprconsent-button').length) {
					// Hide the div if it's visible
					if ($('.gdprconsent-button').is(':visible')) {
						$('.gdprconsent-button button').click();
					}
				}
			});
		}
	});
